\node[main, minimum width=13.4cm, minimum height=2.5cm, align=left, text width=12.8cm] (control) at (6.85,5.2)

  {\textbf{Worker \texttt{sentinald}}\\[2pt]{\footnotesize\color{black!60}PTYs and output logs\\Watcher: output to Events\\Metrics, Commands, Recipes\\SQLite: the source of truth}};
\node[box, minimum width=4.1cm, minimum height=2.2cm, align=left, text width=3.7cm] (wb) at (11.45,1.2)
  {\textbf{Another worker}\\[2pt]{\footnotesize\color{black!60}Same daemon, own data\\Paired with its own\\revocable credential}};
