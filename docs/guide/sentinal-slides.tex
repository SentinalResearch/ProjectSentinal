\section{What Sentinal is}

\begin{frame}{Long work lives in a terminal you have to sit at}{Sentinal moves it into Sessions the daemon owns}
\begin{columns}[T]
\begin{column}{0.31\textwidth}\ruled{SEE}{What is running, waiting or failed on every Machine.}\end{column}
\begin{column}{0.31\textwidth}\ruled{ANSWER}{Get a push when a run fails or waits. Answer an agent's question from the dashboard.}\end{column}
\begin{column}{0.31\textwidth}\ruled{DRIVE}{Open the live terminal on a phone and type into it.}\end{column}
\end{columns}
\vfill
{\small\usebeamercolor[fg]{framesubtitle}Local-first: a daemon per Machine, data in SQLite on that Machine, traffic only inside Tailscale.}
\end{frame}

\begin{frame}{Every Session has one state}{States are never colour-only: the label is always there}
\vspace{4mm}
\srstate{CREATED}\ \srstate{RUNNING}\ \srstate{WAITING}\ \srstate{IDLE}\ \srstate{STALLED}\par\vspace{3mm}
\srstate{COMPLETED}\ \srstate{FAILED}\ \srstate{INTERRUPTED}\par
\vspace{8mm}
\begin{columns}[T]
\begin{column}{0.47\textwidth}\ruled{WAITING}{A prompt or an agent's question needs an answer.}\end{column}
\begin{column}{0.47\textwidth}\ruled{STALLED}{Quiet for ten minutes while it should be working.}\end{column}

\begin{frame}[fragile]{Four ways work enters Sentinal}{From full terminals to observing a process you already started}
\begin{columns}[T]
\begin{column}{0.23\textwidth}\ruled{FULL}{\texttt{sentinal watch}\par\small A persistent terminal: output, screen, input.}\end{column}
\begin{column}{0.23\textwidth}\ruled{COMMANDS}{\texttt{sentinal hook}\par\small Commands from ordinary terminals.}\end{column}
\begin{column}{0.23\textwidth}\ruled{AUTO}{\texttt{sentinal auto on}\par\small Every new terminal starts inside a Session.}\end{column}
\begin{column}{0.23\textwidth}\ruled{PROCESS}{\texttt{sentinal adopt}\par\small Metrics and exit of a running PID. Read-only.}\end{column}
\end{columns}
\end{frame}

\section{The watcher}

\begin{frame}{Output becomes Events, with no model}{Deterministic: the same input always gives the same Events}
\begin{columns}[T]
\begin{column}{0.18\textwidth}\ruled{1 FRAME}{\small Bytes to bounded lines, escapes removed.}\end{column}
\begin{column}{0.18\textwidth}\ruled{2 OBSERVE}{\small Lines, Commands, processes, metrics, screens.}\end{column}
\begin{column}{0.18\textwidth}\ruled{3 ADAPT}{\small Generic, training, pytest, cargo, servers, agents.}\end{column}
\begin{column}{0.18\textwidth}\ruled{4 LIMIT}{\small Progress at most every 5 s.}\end{column}
\begin{column}{0.18\textwidth}\ruled{5 HEALTH}{\small Healthy, warning, stalled or failed, with evidence.}\end{column}
\end{columns}
\end{frame}

\begin{frame}{Sentinal reads the agent's screen}{Read as lines, questions were garbled, missed, or ended early}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\ruled{DETECT}{\small A numbered menu of 2 to 6 options and the question above it.}
\vspace{2mm}
\ruled{END}{\small When the menu leaves the screen. No time limit.}
\end{column}
\begin{column}{0.47\textwidth}
\ruled{WORKS WITH}{\small Any program with a numbered menu: Claude Code, Codex, Vibe, Antigravity and more.}
\vspace{2mm}
\ruled{ANSWER}{\small Click or tap an option. The input is audit-logged.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Progress without changing the program}{Explicit counters only: a bare fraction is never progress}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize\color{srNightText},aboveskip=0.4em,belowskip=0.4em]
Evaluating:  42%|####      | 21/50 [00:10<00:14]   -> 21 of 50 item
[4/10] building                                    -> 4 of 10 item
Processing sample 12 of 200                        -> 12 of 200 sample
accuracy 41/50 on condition A                      -> nothing
\end{lstlisting}
\vspace{2mm}
{\srlabel{NEXT}\par\small Programs being written now can say it themselves, in one printed line:}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize\color{srNightText},aboveskip=0.4em,belowskip=0.4em]
::sentinal progress current=3 total=50 unit=task
\end{lstlisting}
\end{frame}

\section{Daily use}

\begin{frame}{Finished Sessions no longer pile up}{Hide, delete, or clear a whole Machine}
\begin{columns}[T]
\begin{column}{0.31\textwidth}\ruled{HIDE}{\small Leaves every list on every device. History stays searchable; restore from the Hidden page.}\end{column}
\begin{column}{0.31\textwidth}\ruled{DELETE}{\small Output, Commands, Events, metrics and search rows erased. Logged; IDs never reused.}\end{column}
\begin{column}{0.31\textwidth}\ruled{CLEAR FINISHED}{\small One action on the Machine page hides all of its finished Sessions.}\end{column}
\end{columns}
\end{frame}

\begin{frame}{Navigation that takes no space}{The page name drops down the main places}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\ruled{PLACES}{Overview, Terminals, Machines, History, Recipes, Settings.}
\end{column}
\begin{column}{0.47\textwidth}
\ruled{OPENS}{On hover on a desktop, on a tap on a phone, with Enter from the keyboard. Each place shows its prompt word.}
\end{column}
\end{columns}
\end{frame}

\section{Development cycle}

\begin{frame}{Development never touches production}{After a test daemon replaced this Mac's Sessions on the control, 5 November}
\begin{columns}[T]
\begin{column}{0.31\textwidth}\ruled{JUST DEV}{\small Own data, config, ports 5273 and 7520. A DEV badge.}\end{column}
\begin{column}{0.31\textwidth}\ruled{ISOLATED}{\small Refuses production paths. Never links, never updates itself.}\end{column}
\begin{column}{0.31\textwidth}\ruled{ONE INSTALL}{\small Install ID and build commit in every heartbeat; the control refuses a second live install.}\end{column}
\end{columns}
\end{frame}
